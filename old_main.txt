import RPi.GPIO as GPIO
import serial
import subprocess
import time
import sys

# ── Bluetooth 설정 ──────────────────────────────────────────
#BT_ADDR_1    = "20:14:12:02:06:16"  # open/close 제어
BT_ADDR_2    = "50:8C:B1:69:D4:46"  # 방향 제어

#RFCOMM_DEV_1 = "/dev/rfcomm0" D8:A9:8B:AE:AE:86
RFCOMM_DEV_2 = "/dev/rfcomm1"
RFCOMM_CH    = 1  # HC-06 기본 채널, 변경 시 AT+UART로 확인

def bind_rfcomm(dev, addr):
    try:
        result = subprocess.run(
            ["sudo", "rfcomm", "bind", dev, addr, str(RFCOMM_CH)],
            capture_output=True, text=True
        )
        if result.returncode == 0:
            print(f"[BT] {dev} 바인딩 성공 ({addr})")
        else:
            stderr = result.stderr.strip()
            if "already" in stderr.lower():
                print(f"[BT] {dev} 이미 바인딩되어 있음 — 스킵")
            else:
                print(f"[BT] 바인딩 실패 ({dev}): {stderr}", file=sys.stderr)
                sys.exit(1)
    except FileNotFoundError:
        print("[BT] rfcomm 명령어를 찾을 수 없습니다. bluez 패키지를 확인하세요.", file=sys.stderr)
        sys.exit(1)

# ── GPIO 설정 ───────────────────────────────────────────────
RELAY1 = 2
RELAY2 = 3

state = False

def setup_gpio():
    GPIO.setmode(GPIO.BCM)
    GPIO.setup(RELAY1, GPIO.OUT)
    GPIO.setup(RELAY2, GPIO.OUT)

def apply_state():
    level = GPIO.HIGH if state else GPIO.LOW  # Active High 기준
    GPIO.output(RELAY1, level)
    GPIO.output(RELAY2, level)

def set_on():
    global state
    state = True
    apply_state()
    print("[GPIO] 릴레이 ON")

def set_off():
    global state
    state = False
    apply_state()
    print("[GPIO] 릴레이 OFF")

# ── Serial 명령 ─────────────────────────────────────────────
def open_serial(dev, baud=9600, retries=3, delay=1.0):
    for attempt in range(1, retries + 1):
        try:
            s = serial.Serial(dev, baud, timeout=2)
            print(f"[Serial] {dev} 연결 성공")
            return s
        except serial.SerialException as e:
            print(f"[Serial] 연결 시도 {attempt}/{retries} 실패 ({dev}): {e}")
            if attempt < retries:
                time.sleep(delay)
    print(f"[Serial] {dev} 연결에 최종 실패했습니다.", file=sys.stderr)
    sys.exit(1)

def send_cmd(ser, data: bytes, label: str = ""):
    try:
        ser.write(data)
        tag = f" → {label}" if label else ""
        print(f"[BT]{tag} '{data.decode().strip()}' 전송")
    except serial.SerialException as e:
        print(f"[Serial] 전송 오류: {e}", file=sys.stderr)
